# -------------------------------------------------------------------------
# 3. CORE FLASK ROUTES FOR FRONTEND SYNC
# -------------------------------------------------------------------------
@app.route('/')
def index():
    return render_template('index.html', groups=GROUPS, current_group=current_group)

@app.route('/api/groups')
def get_groups():
    """Returns group list and associated candidate vocabulary for UI render."""
    formatted_groups = {}
    for gid, gdata in GROUPS.items():
        if isinstance(gdata, dict) and 'classes' in gdata:
            formatted_groups[gid] = gdata['classes']
        elif isinstance(gdata, list):
            formatted_groups[gid] = gdata
        else:
            formatted_groups[gid] = []
    return jsonify(formatted_groups)

@app.route('/api/group', methods=['GET', 'POST'])
def handle_group():
    """Handles switching sign model groups via dropdown."""
    global current_group
    if request.method == 'POST':
        data = request.get_json(force=True, silent=True) or {}
        group_id = data.get('group_id') or data.get('group')
        
        if group_id in GROUPS:
            with state_lock:
                current_group = group_id
                latest_detection_state['group_id'] = group_id
            
            ginfo = GROUPS[group_id]
            classes = ginfo.get('classes', ginfo) if isinstance(ginfo, dict) else ginfo
            return jsonify({'status': 'success', 'group': group_id, 'classes': classes})
        return jsonify({'status': 'error', 'message': f'Invalid group: {group_id}'}), 400
    
    with state_lock:
        ginfo = GROUPS[current_group]
        classes = ginfo.get('classes', ginfo) if isinstance(ginfo, dict) else ginfo
        return jsonify({'current_group': current_group, 'classes': classes})
